/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public boolean isSymmetric(TreeNode root) {
        return checkMirror(root.left, root.right);
    }

    private static boolean checkMirror(TreeNode left, TreeNode right){

        // Checking both node are null or not
        if(left == null && right == null){
            return true;
        }
        // Checking on node is null or not
        if(left == null || right == null){
            return false;
        }

        // Checking Value is equal or not
        if(left.val != right.val){
            return false;
        }
        // Checking Mirror Position using Recursion and Return
        return checkMirror(left.left, right.right) && checkMirror(left.right, right.left);
    }
}
