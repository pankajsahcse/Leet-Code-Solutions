class Solution {
    public int removeElement(int[] nums, int val) {
        int index = 0;
        for (int i =0; i< nums.length; i++){
            // If our number is not matching with value then, add that value in same array
            if(nums[i] != val){
                nums[index] = nums[i];
                index++;
            }
        }
        return index;
    }
}
