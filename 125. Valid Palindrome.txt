class Solution {
    public boolean isPalindrome(String s) {
        // Remove all non-alphanumeric characters and convert into lowercase 
        s = s.replaceAll("[^a-zA-Z0-9]", "").toLowerCase();
        
        int i = 0;
        int j = s.length()-1;

        while(i<j){
            // checking 1st char with last char 
            if(s.charAt(i) != s.charAt(j)){
                return false;
            }
            i++;
            j--;
        }
        return true;
    }
}
