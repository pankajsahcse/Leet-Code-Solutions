
class Solution {
    public int searchInsert(int[] nums, int target) {
        int start = 0;
        int end = nums.length - 1;

        while (start <= end) {
            int mid = start + (end - start) / 2;

            if (nums[mid] == target) {
                // if target is matched return that index
                return mid;
            } 
            else if (target < nums[mid]) {
                // if target is less than middle of the element then 
                end = mid - 1;
            } 
            else {               
                // if target is greater than middle of the element then 
                start = mid + 1;
            }
        }
        // If target not found then target should be here
        return start;
    }
}
