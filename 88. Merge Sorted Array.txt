class Solution {
    public void merge(int[] nums1, int m, int[] nums2, int n) {
        int i = m -1;
        int j = n -1; 
        int k = m + n -1; // position for last index of nums1, becouse nums1 is bigger than nums2

 // 1st while untill, jb tk i and j ki value compare ho rhi
 // 2nd while , jb j me value hai to nums1 me fill kr do

        while(i >=0 && j >= 0){
            if(nums1[i] > nums2[j]){
                nums1[k] = nums1[i]; // bigger value at last position
                i--;
            }else{
                nums1[k] = nums2[j];
                j--;
            }
            k--;
        }
        // run this loop untill j not be zero becouse j is less than i
        while(j >= 0){
            nums1[k] = nums2[j];
            j--;
            k--;
        }
    }
}
