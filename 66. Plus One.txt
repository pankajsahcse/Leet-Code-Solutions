class Solution {
    public int[] plusOne(int[] digits) {
        int n = digits.length;
        for (int i = n - 1; i >= 0; i--) {
            // checking last digit is 9 or not, if not 9 then increase their value by 1
            if (digits[i] != 9) {
                digits[i]++;
                break;
            } else {
                // if i = 9 then at that index put 0 
                digits[i] = 0;
            }
        }
        if (digits[0] == 0) {
            // if 0 at the 1st index then create new array (size + 1) and put 1 at 1st position
            int[] res = new int[n + 1];
            res[0] = 1;
            return res;
        }
        return digits;
    }
}
