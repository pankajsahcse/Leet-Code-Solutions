class Solution {
    public int lengthOfLastWord(String s) {
    //    if(s.length() == 0) return -1;
        int count = 0;
        String [] splitWord =  s.split(" ");
        int length = splitWord.length;
        String lastWord = splitWord[length - 1];
        for(int i = 0; i < lastWord.length(); i++){
            count++;
        }
        return count;
    }
}
