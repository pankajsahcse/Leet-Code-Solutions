class Solution {
    public int singleNumber(int[] nums) {
        HashSet<Integer> set = new HashSet<>();

        for(int num : nums){
            if(set.contains(num)){ // checking the value is available or not in set
                set.remove(num); // Remove if already present
            }else{
                set.add(num); // Add if not present
            }
        }
        return set.iterator().next();
    }
}
